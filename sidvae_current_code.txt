//==============================================================================
void SIDVaeAudioProcessorEditor::WaveformView::setTable (const std::vector<float>& t, bool vae)
{
    table   = t;
    vaeMode = vae;
    repaint();
}

void SIDVaeAudioProcessorEditor::WaveformView::paint (juce::Graphics& g)
{
    auto r = getLocalBounds().toFloat();
    g.setColour (kPanel);
    g.fillRoundedRectangle (r, 6.0f);

    r = r.reduced (10.0f, 12.0f);
    g.setColour (kDim.withAlpha (0.35f));
    g.drawHorizontalLine ((int) r.getCentreY(), r.getX(), r.getRight());

    if (table.size() < 2)
        return;

    juce::Path path;
    const int   n    = (int) table.size();
    const int   step = juce::jmax (1, n / (int) juce::jmax (1.0f, r.getWidth() * 2.0f));
    for (int i = 0; i <= n; i += step)
    {
        const float v = table[(size_t) (i % n)];
        const float x = r.getX() + r.getWidth() * (float) i / (float) n;
        const float y = r.getCentreY() - juce::jlimit (-1.0f, 1.0f, v) * r.getHeight() * 0.5f;
        if (i == 0) path.startNewSubPath (x, y);
        else        path.lineTo (x, y);
    }

    g.setColour (vaeMode ? kVae : kFallback);
    g.strokePath (path, juce::PathStrokeType (1.8f, juce::PathStrokeType::curved));
}

void SIDVaeAudioProcessorEditor::setupChoice (Choice& c, const juce::String& paramId, const juce::String& text)
{
    // Items must exist before the attachment is created.
    if (auto* param = dynamic_cast<juce::AudioParameterChoice*> (proc.apvts.getParameter (paramId)))
        c.box.addItemList (param->choices, 1);
    addAndMakeVisible (c.box);

    c.label.setText (text, juce::dontSendNotification);
    c.label.setJustificationType (juce::Justification::centred);
    c.label.setColour (juce::Label::textColourId, kDim);
    addAndMakeVisible (c.label);

    c.attachment = std::make_unique<juce::AudioProcessorValueTreeState::ComboBoxAttachment> (proc.apvts, paramId, c.box);
}

    status.setText (proc.getStatusText(), juce::dontSendNotification);

    waveView.setTable (scratch, vae);

{
    auto r = getLocalBounds().reduced (12, 8);

    auto header = r.removeFromTop (28);
    header.removeFromLeft (120);
    status.setBounds (header);

    r.removeFromTop (6);
    waveView.setBounds (r.removeFromTop (130));
    r.removeFromTop (10);

    const int rowH = r.getHeight() / 2;
    auto row1 = r.removeFromTop (rowH);
    auto row2 = r;
    const int w = row1.getWidth() / 5;

    layoutChoice (waveA, row1.removeFromLeft (w));
    layoutKnob   (morph, row1.removeFromLeft (w));
    layoutChoice (waveB, row1.removeFromLeft (w));
    layoutKnob   (pw,    row1.removeFromLeft (w));
    layoutKnob   (vel,   row1);

    layoutKnob   (z1,    row2.removeFromLeft (w));
    layoutKnob   (z2,    row2.removeFromLeft (w));
    layoutKnob   (z3,    row2.removeFromLeft (w));
    layoutKnob   (freq,  row2.removeFromLeft (w));
    layoutKnob   (out,   row2);
}

// Model discovery + metadata
//==============================================================================
namespace
{
    const juce::String kDefaultModelFile = "sid_vae_decoder.onnx";
    const juce::String kMetaFile         = "sid_vae_meta.json";

    // Where the decoder is looked for, in order.
    juce::Array<juce::File> candidateModelDirs()
    {
        juce::Array<juce::File> dirs;

        // 1. Explicit override, handy while experimenting with different models.
        const auto env = juce::SystemStats::getEnvironmentVariable ("SIDVAE_MODEL_DIR", {});
        if (env.isNotEmpty() && juce::File::isAbsolutePath (env))
            dirs.add (juce::File (env));

        // 2. Inside the plugin bundle (CMake copies Models/ here).
        //    macOS: <bundle>/Contents/MacOS/<binary>  ->  <bundle>/Contents/Resources/Models
        const auto exeDir = juce::File::getSpecialLocation (juce::File::currentExecutableFile).getParentDirectory();
        dirs.add (exeDir.getParentDirectory().getChildFile ("Resources").getChildFile ("Models"));
        dirs.add (exeDir.getChildFile ("Models"));

        // 3. Per-user install location.
        auto userData = juce::File::getSpecialLocation (juce::File::userApplicationDataDirectory);
       #if JUCE_MAC
        userData = userData.getChildFile ("Application Support");
       #endif
        dirs.add (userData.getChildFile ("SIDVae").getChildFile ("Models"));

        // 4. The source tree's Models/ folder (development builds).
       #ifdef SIDVAE_DEV_MODEL_DIR
        dirs.add (juce::File (SIDVAE_DEV_MODEL_DIR));
       #endif
        return dirs;
    }

    struct ModelMeta
    {
        int latentDim = 3;
        int condDim   = 6;
        std::vector<std::string> condOrder = sidvae::defaultCondOrder();
        sidvae::LogF0Norm f0Norm;
        juce::String decoderFile = kDefaultModelFile;
    };

    // Reads sid_vae_meta.json. Missing optional keys keep their defaults.
    bool parseMeta (const juce::File& file, ModelMeta& m, juce::String& error)
    {
        const auto json = juce::JSON::parse (file);
        if (! json.isObject())
        {
            error = "could not parse " + file.getFileName();
            return false;
        }

        if (json.hasProperty ("latent_dim"))   m.latentDim   = (int) json["latent_dim"];
        if (json.hasProperty ("cond_dim"))     m.condDim     = (int) json["cond_dim"];
        if (json.hasProperty ("decoder_file")) m.decoderFile = json["decoder_file"].toString();

        if (auto* arr = json["cond_order"].getArray())
        {
            m.condOrder.clear();
            for (const auto& v : *arr)
                m.condOrder.push_back (v.toString().toStdString());
        }

        if (json.hasProperty ("log_f0_mean") && json.hasProperty ("log_f0_std"))
        {
            m.f0Norm.enabled = true;
            m.f0Norm.mean    = (float) (double) json["log_f0_mean"];
            m.f0Norm.std     = (float) (double) json["log_f0_std"];
        }

        if ((int) m.condOrder.size() != m.condDim)
        {
            error = "cond_order has " + juce::String ((int) m.condOrder.size())
                  + " names but cond_dim is " + juce::String (m.condDim);
            return false;
        }
        if (m.latentDim <= 0 || m.condDim <= 0)
        {
            error = "latent_dim / cond_dim must be positive";
            return false;
        }
        return true;
    }
}

//==============================================================================
// TableBuilder: owns the decoder, watches the parameters, publishes tables.
// Runs on its own thread so ONNX inference never touches the audio thread.
//==============================================================================
class SIDVaeAudioProcessor::TableBuilder : public juce::Thread
{
public:
    explicit TableBuilder (SIDVaeAudioProcessor& p)
        : juce::Thread ("SIDVae table builder"), proc (p) {}

    ~TableBuilder() override { stopThread (4000); }

    void run() override
    {
        loadModel();

        Snapshot last;
        bool haveLast = false;

        while (! threadShouldExit())
        {
            const bool forced = proc.rebuildRequested.exchange (false);
            const Snapshot s  = readParams();

            if (forced || ! haveLast || ! s.sameAs (last, decoder.isReady()))
            {
                float* dst = proc.tables.writeBuffer();
                buildTable (s, dst);
                {
                    const juce::ScopedLock sl (proc.infoLock);
                    std::copy (dst, dst + kTableLen, proc.displayTable.begin());
                }
                proc.tables.publish();
                last = s;
                haveLast = true;
            }

            wait (5);   // ~200 Hz control rate; decodes only when something changed
        }
    }

private:
    struct Snapshot
    {
        int   waveA = 0, waveB = 1;
        float morph = 0.0f, pw = 0.5f, vel = 1.0f, freq = 220.0f;
        float z[3] = { 0.0f, 0.0f, 0.0f };

        bool sameAs (const Snapshot& o, bool freqMatters) const
        {
            auto close = [] (float a, float b) { return std::abs (a - b) < 1.0e-5f; };
            if (waveA != o.waveA || waveB != o.waveB) return false;
            if (! close (morph, o.morph) || ! close (pw, o.pw) || ! close (vel, o.vel)) return false;
            for (int i = 0; i < 3; ++i) if (! close (z[i], o.z[i])) return false;
            // f0 is a decoder input (log_f0), so pitch changes re-decode in VAE mode.
            if (freqMatters && std::abs (freq - o.freq) > 0.001f * o.freq) return false;
            return true;
        }
    };

    Snapshot readParams() const
    {
        Snapshot s;
        s.waveA = juce::jlimit (0, sidvae::kNumWaves - 1, (int) proc.pWaveA->load());
        s.waveB = juce::jlimit (0, sidvae::kNumWaves - 1, (int) proc.pWaveB->load());
        s.morph = juce::jlimit (0.0f, 1.0f, proc.pMorph->load());
        s.pw    = proc.pPw->load();
        s.vel   = proc.pVel->load();
        s.freq  = proc.pFreq->load();
        for (int i = 0; i < 3; ++i) s.z[i] = proc.pZ[i]->load();
        return s;
    }

    void loadModel()
    {
        juce::File modelDir;
        for (const auto& dir : candidateModelDirs())
        {
            if (dir.getChildFile (kDefaultModelFile).existsAsFile()
                || dir.getChildFile (kMetaFile).existsAsFile())
            {
                modelDir = dir;
                break;
            }
        }

        if (modelDir == juce::File())
        {
            loadError = "no " + kDefaultModelFile + " found (see README: Model files)";
            refreshStatus();
            return;
        }

        // Meta next to the model, or one level up 
        auto metaFile = modelDir.getChildFile (kMetaFile);
        if (! metaFile.existsAsFile())
            metaFile = modelDir.getParentDirectory().getChildFile (kMetaFile);

        if (metaFile.existsAsFile())
        {
            juce::String metaErr;
            if (! parseMeta (metaFile, meta, metaErr))
            {
                loadError = metaErr;
                refreshStatus();
                return;
            }
        }
        else
        {
            metaNote = "no " + kMetaFile + ", using defaults";
        }

        const auto modelFile = modelDir.getChildFile (meta.decoderFile);
        const sidvae::SidVaeDecoder::Config cfg { meta.latentDim, meta.condDim, kTableLen };

        std::string err;
        if (! decoder.load (modelFile.getFullPathName().toStdString(), cfg, err))
        {
            loadError = juce::String (err);
            refreshStatus();
            return;
        }

        // Warn about any conditioning names this build doesn't know how to fill.
        std::vector<std::string> unknown;
        std::vector<float> scratch (meta.condOrder.size());
        sidvae::buildCondition ({}, meta.condOrder, meta.f0Norm, scratch.data(), &unknown);
        for (const auto& u : unknown) unknownCond.add (juce::String (u));

        if (meta.latentDim != 3)
            metaNote = "latent_dim is " + juce::String (meta.latentDim) + "; only the first 3 have knobs";

        modelInfo = modelFile.getFileName() + " | " + juce::String (decoder.ioSummary());
        refreshStatus();
    }

    void buildTable (const Snapshot& s, float* dst)
    {
        // Waveform mix at this morph position: one-hot at the poles.
        float w[sidvae::kNumWaves] = { 0.0f, 0.0f, 0.0f };
        w[s.waveA] += 1.0f - s.morph;
        w[s.waveB] += s.morph;

        if (decoder.isReady())
        {
            sidvae::SidControls c;
            c.pw = s.pw;  c.f0Hz = s.freq;  c.vel = s.vel;
            c.waveTri = w[0];  c.waveSaw = w[1];  c.wavePulse = w[2];

            cond.resize (meta.condOrder.size());
            sidvae::buildCondition (c, meta.condOrder, meta.f0Norm, cond.data());

            z.assign ((std::size_t) meta.latentDim, 0.0f);
            for (int i = 0; i < std::min (3, meta.latentDim); ++i) z[(std::size_t) i] = s.z[i];

            std::string err;
            if (decoder.decode (z.data(), cond.data(), decoded, &err))
            {
                sidvae::removeDC   (decoded.data(), decoded.size());
                sidvae::limitPeak  (decoded.data(), decoded.size(), 1.0f);
                std::copy (decoded.begin(), decoded.end(), dst);
                if (decodeError.isNotEmpty()) { decodeError.clear(); refreshStatus(); }
                return;
            }
            if (decodeError != juce::String (err)) { decodeError = juce::String (err); refreshStatus(); }
        }

        // Fallback: plain weighted sum of reference shapes (shape-domain crossfade).
        ensureFallbackTables (s.pw);
        for (int i = 0; i < kTableLen; ++i)
        {
            const auto k = (std::size_t) i;
            dst[i] = w[0] * fallback[0][k] + w[1] * fallback[1][k] + w[2] * fallback[2][k];
        }
        sidvae::limitPeak (dst, (std::size_t) kTableLen, 1.0f);
    }

    void ensureFallbackTables (float pw)
    {
        if (fallback.empty())
        {
            fallback.push_back (sidvae::makeTriangle (kTableLen));
            fallback.push_back (sidvae::makeSaw (kTableLen));
            fallback.push_back (sidvae::makePulse (kTableLen, pw));
            fallbackPw = pw;
        }
        else if (std::abs (pw - fallbackPw) > 1.0e-4f)
        {
            fallback[2] = sidvae::makePulse (kTableLen, pw);
            fallbackPw = pw;
        }
    }

    void refreshStatus()
    {
        const bool vae = decoder.isReady() && decodeError.isEmpty();
        proc.vaeActive.store (vae);

        juce::String s;
        if (vae)
        {
            s = "VAE  |  " + modelInfo;
            if (unknownCond.size() > 0) s << "  |  unknown cond: " << unknownCond.joinIntoString (", ");
        }
        else
        {
            s = "FALLBACK: plain crossfade, no VAE  |  "
              + (decodeError.isNotEmpty() ? "decode failed: " + decodeError : loadError);
        }
        if (metaNote.isNotEmpty()) s << "  |  " << metaNote;

        proc.setStatus (s);
    }

    SIDVaeAudioProcessor& proc;

    sidvae::SidVaeDecoder decoder;
    ModelMeta meta;
    juce::String modelInfo, loadError, decodeError, metaNote;
    juce::StringArray unknownCond;

    std::vector<float> z, cond, decoded;
    std::vector<std::vector<float>> fallback;
    float fallbackPw = -1.0f;
};

//==============================================================================
SIDVaeAudioProcessor::APVTS::ParameterLayout SIDVaeAudioProcessor::createParameterLayout()
{
    using namespace juce;
    std::vector<std::unique_ptr<RangedAudioParameter>> p;

    auto decimals = [] (int places, const String& label = {})
    {
        return AudioParameterFloatAttributes()
            .withLabel (label)
            .withStringFromValueFunction ([places] (float v, int) { return String (v, places); });
    };

    // Index order must match sidvae::Wave (Triangle, Saw, Pulse).
    const StringArray waves { "Triangle", "Saw", "Pulse" };
    p.push_back (std::make_unique<AudioParameterChoice> (ParameterID { "waveA", 1 }, "Wave A", waves, 0));
    p.push_back (std::make_unique<AudioParameterChoice> (ParameterID { "waveB", 1 }, "Wave B", waves, 1));

    p.push_back (std::make_unique<AudioParameterFloat> (ParameterID { "morph", 1 }, "Morph",
                                                        NormalisableRange<float> (0.0f, 1.0f), 0.0f, decimals (2)));
    p.push_back (std::make_unique<AudioParameterFloat> (ParameterID { "pw", 1 }, "Pulse Width",
                                                        NormalisableRange<float> (0.02f, 0.98f), 0.5f, decimals (2)));

    for (int i = 1; i <= 3; ++i)

#pragma once
// Band-limited reference waveforms used ONLY when the VAE is unavailable.
// In that mode the plugin crossfades these in the shape domain, and the UI says so.

#include <cmath>
#include <vector>
#include "TableUtils.h"

namespace sidvae
{
    enum class Wave : int { Triangle = 0, Saw = 1, Pulse = 2 };
    constexpr int kNumWaves = 3;

    namespace detail
    {
        constexpr double kPi    = 3.14159265358979323846;
        constexpr double kTwoPi = 2.0 * kPi;
    }

    inline std::vector<float> makeTriangle (int n, int harmonics = 64)
    {
        std::vector<float> t ((std::size_t) n, 0.0f);
        for (int i = 0; i < n; ++i)
        {
            const double p = (double) i / (double) n;
            double s = 0.0;
            for (int k = 0; k < harmonics / 2; ++k)
            {
                const int h = 2 * k + 1;
                s += ((k % 2 == 0) ? 1.0 : -1.0) * std::sin (detail::kTwoPi * h * p) / (double) (h * h);
            }
            t[(std::size_t) i] = (float) (8.0 / (detail::kPi * detail::kPi) * s);
        }
        return t;
    }

    inline std::vector<float> makeSaw (int n, int harmonics = 64)
    {
        std::vector<float> t ((std::size_t) n, 0.0f);
        for (int i = 0; i < n; ++i)
        {
            const double p = (double) i / (double) n;
            double s = 0.0;
            for (int h = 1; h <= harmonics; ++h)
                s += std::sin (detail::kTwoPi * h * p) / (double) h;
            t[(std::size_t) i] = (float) (-2.0 / detail::kPi * s); // rising ramp
        }
        const float pk = peakAbs (t.data(), t.size());   // Gibbs overshoot -> normalise to 1
        if (pk > 0.0f)
            for (auto& v : t) v /= pk;
        return t;
    }

    // Pulse with duty cycle `pw` (0..1), DC removed, peak-normalised to 1.
    inline std::vector<float> makePulse (int n, float pw, int harmonics = 64)
    {
        const double d = std::fmin (0.98, std::fmax (0.02, (double) pw));
        std::vector<float> t ((std::size_t) n, 0.0f);
        for (int i = 0; i < n; ++i)
        {
            const double p = (double) i / (double) n;
            double s = 0.0;
            for (int h = 1; h <= harmonics; ++h)
                s += (2.0 / (detail::kPi * h)) * std::sin (detail::kPi * h * d)
                     * std::cos (detail::kTwoPi * h * (p - 0.5 * d));
            t[(std::size_t) i] = (float) s;
        }
        removeDC (t.data(), t.size());
        const float pk = peakAbs (t.data(), t.size());
        if (pk > 0.0f)
            for (auto& v : t) v /= pk;
        return t;
    }
}

#pragma once
// Builds the decoder's conditioning vector by NAME, using `cond_order` from
// sid_vae_meta.json, so the C++ side can't silently drift from the training order.
//
//   pw_norm    : pulse width, 0..1
//   log_f0     : natural log of f0 in Hz; if the meta has "log_f0_mean" and
//                "log_f0_std" it is standardised as (ln f0 - mean) / std
//   vel01      : velocity, 0..1
//   wave_tri / wave_saw / wave_pulse : waveform flags (one-hot at the poles)

#include <cmath>
#include <string>
#include <vector>

namespace sidvae
{
    struct SidControls
    {
        float pw      = 0.5f;
        float f0Hz    = 220.0f;
        float vel     = 1.0f;
        float waveTri = 1.0f, waveSaw = 0.0f, wavePulse = 0.0f;
    };

    struct LogF0Norm
    {
        bool  enabled = false;
        float mean = 0.0f, std = 1.0f;
    };

    inline std::vector<std::string> defaultCondOrder()
    {
        return { "pw_norm", "log_f0", "vel01", "wave_tri", "wave_saw", "wave_pulse" };
    }

    // Writes order.size() values into `out`. Names it doesn't recognise are set to 0
    // and appended to `unknown` (if given) so the UI can warn about them.
    inline void buildCondition (const SidControls& c,
                                const std::vector<std::string>& order,
                                const LogF0Norm& norm,
                                float* out,
                                std::vector<std::string>* unknown = nullptr)
    {
        for (std::size_t i = 0; i < order.size(); ++i)
        {
            const std::string& k = order[i];
            float v = 0.0f;

            if      (k == "pw_norm")    v = c.pw;
            else if (k == "vel01")      v = c.vel;
            else if (k == "wave_tri")   v = c.waveTri;
            else if (k == "wave_saw")   v = c.waveSaw;
            else if (k == "wave_pulse") v = c.wavePulse;
            else if (k == "log_f0")
            {
                v = std::log (std::fmax (1.0f, c.f0Hz));
                if (norm.enabled && std::abs (norm.std) > 1.0e-12f)
                    v = (v - norm.mean) / norm.std;
            }
            else if (unknown != nullptr)
                unknown->push_back (k);

            out[i] = v;
        }
    }
}

==> Source/core/TableUtils.h <==
#pragma once
// Small, allocation-free helpers for single-cycle wavetables. No JUCE dependency.

#include <algorithm>
#include <cmath>
#include <cstddef>

namespace sidvae
{
    // Linear resampling of one *periodic* cycle: wraps at the end instead of clamping,
    // so the last sample interpolates back towards the first.
    inline void resampleCyclic (const float* src, std::size_t srcLen, float* dst, std::size_t dstLen) noexcept
    {
        if (srcLen == 0 || dstLen == 0)
            return;

        const double step = (double) srcLen / (double) dstLen;
        for (std::size_t i = 0; i < dstLen; ++i)
        {
            const double pos  = (double) i * step;
            const auto   i0   = (std::size_t) pos % srcLen;
            const auto   i1   = (i0 + 1) % srcLen;
            const float  frac = (float) (pos - std::floor (pos));
            dst[i] = src[i0] + frac * (src[i1] - src[i0]);
        }
    }

    // A wavetable oscillator turns any DC in the table into a constant offset, remove it.
    inline void removeDC (float* t, std::size_t n) noexcept
    {
        if (n == 0) return;
        double sum = 0.0;
        for (std::size_t i = 0; i < n; ++i) sum += t[i];
        const float mean = (float) (sum / (double) n);
        for (std::size_t i = 0; i < n; ++i) t[i] -= mean;
    }

    inline float peakAbs (const float* t, std::size_t n) noexcept
    {
        float p = 0.0f;
        for (std::size_t i = 0; i < n; ++i) p = std::max (p, std::abs (t[i]));
        return p;
    }

    // Only scales *down*: keeps whatever level differences the model produces,
    // but stops a bad decode from blasting the output.
    inline void limitPeak (float* t, std::size_t n, float maxPeak = 1.0f) noexcept
    {
        const float p = peakAbs (t, n);
        if (p > maxPeak && p > 0.0f)
        {
            const float g = maxPeak / p;
            for (std::size_t i = 0; i < n; ++i) t[i] *= g;
        }
    }

    inline bool allFinite (const float* t, std::size_t n) noexcept
    {
        for (std::size_t i = 0; i < n; ++i)
            if (! std::isfinite (t[i])) return false;
        return true;
    }
}

==> Source/core/TripleBuffer.h <==
#pragma once
// Lock-free single-producer / single-consumer triple buffer for wavetables.
// The decode thread writes into its own buffer and publishes; the audio thread
// picks up the newest complete table at the start of a block. Neither side ever
// waits on the other, and nothing allocates after prepare().

#include <array>
#include <atomic>
#include <vector>

namespace sidvae
{
    class TableTripleBuffer
    {
    public:
        // Call before either thread starts using the buffer.
        void prepare (int tableLen)
        {
            for (auto& b : bufs)
                b.assign ((std::size_t) tableLen, 0.0f);
            front = 0;
            middle.store (1);
            back = 2;
        }

        int size() const noexcept { return (int) bufs[0].size(); }

        // ---- producer (decode thread) ----
        float* writeBuffer() noexcept { return bufs[(std::size_t) back].data(); }

        void publish() noexcept
        {
            back = middle.exchange (back | kDirty, std::memory_order_acq_rel) & kIndexMask;
        }

        // ---- consumer (audio thread) ----
        // Returns true if a newer table was swapped in since the last call.
        bool acquire() noexcept
        {
            if ((middle.load (std::memory_order_acquire) & kDirty) == 0)
                return false;
            front = middle.exchange (front, std::memory_order_acq_rel) & kIndexMask;
            return true;
        }

        const float* readBuffer() const noexcept { return bufs[(std::size_t) front].data(); }

    private:
        static constexpr int kDirty     = 4;
        static constexpr int kIndexMask = 3;

        std::array<std::vector<float>, 3> bufs;
        int front = 0;               // audio thread only
        std::atomic<int> middle { 1 };
        int back = 2;                // decode thread only
    };
}

==> Source/core/WavetableOsc.h <==
#pragma once
// Single-table wavetable oscillator. When a new table arrives it fades from the
// table currently heard to the new one over a few milliseconds, which removes the
// click of a hard swap. This is de-zippering in time, not the timbre morph:
// the morph itself happens in the decoder, before a table ever reaches here.
// No allocation in render(); no JUCE dependency.

#include <algorithm>
#include <cmath>
#include <cstring>
#include <vector>

namespace sidvae
{
    class WavetableOsc
    {
    public:
        void prepare (double sampleRate, int tableLen, double fadeMs = 8.0)
        {
            sr  = sampleRate > 0.0 ? sampleRate : 44100.0;
            len = std::max (16, tableLen);
            cur .assign ((std::size_t) len, 0.0f);
            prev.assign ((std::size_t) len, 0.0f);
            fadeLen  = std::max (1, (int) std::lround (sr * fadeMs * 0.001));
            fadePos  = fadeLen;  // not fading
            phase    = 0.0;
            gainSmoothed = 0.0f;
            gainCoeff = (float) (1.0 - std::exp (-1.0 / (0.010 * sr))); // ~10 ms
        }

        // Audio thread. `src` must hold tableLen samples.
        void setTargetTable (const float* src) noexcept
        {
            // Bake whatever is audible right now into `prev` so a table that lands
            // mid-fade continues from the current mix instead of jumping.
            const float f = fadeAmount();
            if (f < 1.0f)
                for (int i = 0; i < len; ++i)
                    prev[(std::size_t) i] += f * (cur[(std::size_t) i] - prev[(std::size_t) i]);
            else
                std::swap (prev, cur);   // vector swap: pointer exchange, no allocation

            std::memcpy (cur.data(), src, sizeof (float) * (std::size_t) len);
            fadePos = 0;
        }

        void render (float* dst, int numSamples, float freqHz, float gainLinear) noexcept
        {
            const double f   = std::clamp ((double) freqHz, 0.0, 0.49 * sr);
            const double inc = (double) len * f / sr;

            for (int n = 0; n < numSamples; ++n)
            {
                const int   i0   = (int) phase;
                const int   i1   = (i0 + 1 == len) ? 0 : i0 + 1;
                const float frac = (float) (phase - (double) i0);

                const float c = cur[(std::size_t) i0] + frac * (cur[(std::size_t) i1] - cur[(std::size_t) i0]);
                float s = c;
                if (fadePos < fadeLen)
                {
                    const float p = prev[(std::size_t) i0] + frac * (prev[(std::size_t) i1] - prev[(std::size_t) i0]);
                    const float a = (float) fadePos / (float) fadeLen;
                    s = p + a * (c - p);
                    ++fadePos;
                }

                gainSmoothed += gainCoeff * (gainLinear - gainSmoothed);
                dst[n] = s * gainSmoothed;

                phase += inc;
                if (phase >= (double) len) phase -= (double) len;
            }
        }

        // Copy of the table currently being faded towards (for tests / debugging).
        const std::vector<float>& currentTable() const noexcept { return cur; }

    private:
        float fadeAmount() const noexcept { return fadePos >= fadeLen ? 1.0f : (float) fadePos / (float) fadeLen; }

        std::vector<float> cur, prev;
        double sr = 44100.0;
        int    len = 2048;
        int    fadeLen = 1, fadePos = 1;
        double phase = 0.0;
        float  gainSmoothed = 0.0f, gainCoeff = 0.01f;
    };
}

#include <cctype>

 #include <onnxruntime_cxx_api.h>

    Ort::Env& ortEnv()
    {
        static Ort::Env env (ORT_LOGGING_LEVEL_WARNING, "SIDVae");
        return env;
    }

    struct InputInfo
    {
        std::string name;
        std::vector<int64_t> shape;   // as declared; dynamic dims are <= 0
        int64_t perItem = -1;         // elements per batch item, -1 if unknown
    };

    int64_t elementsPerItem (const std::vector<int64_t>& s)
    {
        if (s.empty())      return -1;
        if (s.size() == 1)  return s[0] > 0 ? s[0] : -1;   // no batch dim
        int64_t p = 1;
        for (std::size_t i = 1; i < s.size(); ++i)
        {
            if (s[i] <= 0) return -1;
            p *= s[i];
        }
        return p;
    }

    // Turn a declared shape (possibly with dynamic dims) into a concrete one holding n elements.
    std::vector<int64_t> concreteShape (const std::vector<int64_t>& declared, int64_t n)
    {
        if (declared.size() == 1) return { n };
        if (declared.size() < 2)  return { 1, n };

        std::vector<int64_t> s = declared;
        s[0] = 1;                               // batch of one
        int unknownAt = -1;
        int64_t known = 1;
        for (std::size_t i = 1; i < s.size(); ++i)
        {
            if (s[i] <= 0)
            {
                if (unknownAt >= 0) return { 1, n };
                unknownAt = (int) i;
            }
            else known *= s[i];
        }
        if (unknownAt >= 0)
        {
            if (known == 0 || n % known != 0) return { 1, n };
            s[(std::size_t) unknownAt] = n / known;
        }
        else if (known != n) return { 1, n };
        return s;
    }

    std::string shapeToString (const std::vector<int64_t>& s)
    {
        std::ostringstream os;
        os << '[';
        for (std::size_t i = 0; i < s.size(); ++i) os << (i ? "," : "") << s[i];
        os << ']';
        return os.str();
    }

    bool looksLikeCondition (std::string n)
    {
        std::transform (n.begin(), n.end(), n.begin(), [] (unsigned char ch) { return (char) std::tolower (ch); });
        return n.find ("cond") != std::string::npos || n.find ("label") != std::string::npos || n == "c" || n == "y";
    }

            // Keep TypeInfo alive: the shape info is a view into it.

            if (t.GetElementType() != ONNX_TENSOR_ELEMENT_DATA_TYPE_FLOAT)

            info.shape   = t.GetShape();
            info.perItem = elementsPerItem (info.shape);
            ins.push_back (std::move (info));
        }

        if (im->session->GetOutputCount() < 1)
        {
            error = "model has no outputs";
            return false;
        }
        im->outName = im->session->GetOutputNameAllocated (0, alloc).get();

        const int64_t L = cfg.latentDim, C = cfg.condDim;
        std::ostringstream io;

        if (ins.size() == 2)
        {
            int zi = -1;
            const bool sizesKnown = ins[0].perItem > 0 && ins[1].perItem > 0;

            if (sizesKnown && L != C)
            {
                if      (ins[0].perItem == L && ins[1].perItem == C) zi = 0;
                else if (ins[0].perItem == C && ins[1].perItem == L) zi = 1;
            }
            if (zi < 0)
            {
                const bool c0 = looksLikeCondition (ins[0].name), c1 = looksLikeCondition (ins[1].name);
                zi = (c0 && ! c1) ? 1 : 0;
            }
            const auto& zIn = ins[(std::size_t) zi];
            const auto& cIn = ins[(std::size_t) (1 - zi)];

            if ((zIn.perItem > 0 && zIn.perItem != L) || (cIn.perItem > 0 && cIn.perItem != C))
            {
                error = "input sizes don't match meta: '" + zIn.name + "' " + shapeToString (zIn.shape)
                      + ", '" + cIn.name + "' " + shapeToString (cIn.shape)
                      + " vs latent_dim=" + std::to_string (L) + ", cond_dim=" + std::to_string (C);
                return false;
            }

            im->layout = Impl::Layout::TwoInputs;
            im->zName  = zIn.name;   im->zShape = concreteShape (zIn.shape, L);
            im->cName  = cIn.name;   im->cShape = concreteShape (cIn.shape, C);
            im->zBuf.assign ((std::size_t) L, 0.0f);
            im->cBuf.assign ((std::size_t) C, 0.0f);

            io << im->zName << shapeToString (im->zShape) << " + " << im->cName << shapeToString (im->cShape);
        }
        else if (ins.size() == 1)
        {
            if (ins[0].perItem > 0 && ins[0].perItem != L + C)
            {
                error = "single input '" + ins[0].name + "' " + shapeToString (ins[0].shape)
                      + " is not latent_dim + cond_dim = " + std::to_string (L + C);
                return false;
            }
            im->layout   = Impl::Layout::Concatenated;
            im->catName  = ins[0].name;
            im->catShape = concreteShape (ins[0].shape, L + C);
            im->catBuf.assign ((std::size_t) (L + C), 0.0f);

            io << im->catName << shapeToString (im->catShape) << " (z then cond)";
        }
        else
        {
            error = "expected 1 or 2 model inputs, found " + std::to_string (ins.size());
            return false;
        }

        impl  = std::move (im);
        ready = true;

        // Smoke test so a broken model is caught at load time, not mid-performance.
        std::vector<float> z ((std::size_t) L, 0.0f), cond ((std::size_t) C, 0.0f), out;
        std::string decodeErr;
        if (! decode (z.data(), cond.data(), out, &decodeErr))
        {
            ready = false;
            impl.reset();
            error = "test decode failed: " + decodeErr;
            return false;
        }

        io << " -> " << lastOutputLength << " samples";
        if (lastOutputLength != (std::size_t) cfg.tableLen)
            io << " (resampled to " << cfg.tableLen << ")";
        summary = io.str();
        return true;
    }
    catch (const std::exception& e)
    {
        ready = false;
        impl.reset();
        error = e.what();
        return false;
    }
#else
    (void) modelFile;
    error = "built without ONNX Runtime (SIDVAE_WITH_ONNX=OFF)";
    return false;
#endif
}

bool SidVaeDecoder::decode (const float* z, const float* cond, std::vector<float>& out, std::string* error)
{
#if SIDVAE_WITH_ONNX
    if (! ready || impl == nullptr)
    {
        if (error) *error = "decoder not loaded";
        return false;
    }

    try
    {
        auto& im = *impl;
        const auto L = (std::size_t) cfg.latentDim;
        const auto C = (std::size_t) cfg.condDim;

        std::vector<Ort::Value>  inputs;
        std::vector<const char*> names;

        if (im.layout == Impl::Layout::TwoInputs)
        {
            std::copy (z,    z + L,    im.zBuf.begin());
            std::copy (cond, cond + C, im.cBuf.begin());
            inputs.push_back (Ort::Value::CreateTensor<float> (im.mem, im.zBuf.data(), L, im.zShape.data(), im.zShape.size()));
            inputs.push_back (Ort::Value::CreateTensor<float> (im.mem, im.cBuf.data(), C, im.cShape.data(), im.cShape.size()));
            names = { im.zName.c_str(), im.cName.c_str() };
        }
        else
        {
            std::copy (z,    z + L,    im.catBuf.begin());
            std::copy (cond, cond + C, im.catBuf.begin() + (std::ptrdiff_t) L);
            inputs.push_back (Ort::Value::CreateTensor<float> (im.mem, im.catBuf.data(), L + C, im.catShape.data(), im.catShape.size()));
            names = { im.catName.c_str() };
        }

        const char* outNames[] = { im.outName.c_str() };
        auto result = im.session->Run (Ort::RunOptions { nullptr },
                                       names.data(), inputs.data(), inputs.size(),
                                       outNames, 1);

        const std::size_t n = result[0].GetTensorTypeAndShapeInfo().GetElementCount();
        if (n < 2)
        {
            if (error) *error = "model output has fewer than 2 samples";
            return false;
        }
        lastOutputLength = n;

        const float* data = result[0].GetTensorData<float>();
        out.resize ((std::size_t) cfg.tableLen);
        if (n == out.size()) std::copy (data, data + n, out.begin());
        else                 resampleCyclic (data, n, out.data(), out.size());

        if (! allFinite (out.data(), out.size()))
        {
            if (error) *error = "model produced NaN/Inf";
            return false;
        }
        return true;
    }
    catch (const std::exception& e)
    {
        if (error) *error = e.what();
        return false;
    }
#else
    (void) z; (void) cond; (void) out;
    if (error) *error = "built without ONNX Runtime";
    return false;
#endif
}

#!/usr/bin/env python3
"""
Inspect the SIDVae decoder and measure whether its morph is more than a crossfade.

    pip install onnxruntime numpy            # matplotlib optional, for --plot
    python tools/inspect_model.py Models/sid_vae_decoder.onnx --meta sid_vae_meta.json
    python tools/inspect_model.py Models/sid_vae_decoder.onnx --a tri --b pulse --f0 110 --plot morph.png

It builds the conditioning vector exactly like the plugin (Source/core/SidCondition.h):
waveform flags are one-hot at the poles and linearly interpolated by the morph
amount; log_f0 = ln(f0), standardised if the meta has log_f0_mean/log_f0_std.

What it reports, for t in [0, 1]:
  nonlinearity  ||d(t) - lerp(d(0), d(1), t)|| / ||d(1) - d(0)||
                0 means the decoder is just crossfading its two endpoint shapes.
  spectral      the same measure on harmonic magnitudes (phase-insensitive), which
                is closer to what you hear.
  step size     ||d(t_k+1) - d(t_k)|| between neighbouring morph positions. If one
                step is much bigger than the median, the morph jumps there rather
                than moving smoothly.
"""

import argparse
import json
import math
import sys
from pathlib import Path

import numpy as np
import onnxruntime as ort

DEFAULT_ORDER = ["pw_norm", "log_f0", "vel01", "wave_tri", "wave_saw", "wave_pulse"]
WAVES = {"tri": 0, "saw": 1, "pulse": 2}


def load_meta(path):
    meta = {"latent_dim": 3, "cond_dim": 6, "cond_order": DEFAULT_ORDER}
    if path and Path(path).exists():
        meta.update(json.loads(Path(path).read_text()))
    return meta


def build_cond(meta, pw, f0, vel, weights):
    vals = {
        "pw_norm": pw,
        "vel01": vel,
        "wave_tri": weights[0],
        "wave_saw": weights[1],
        "wave_pulse": weights[2],
    }
    lf0 = math.log(max(1.0, f0))
    if "log_f0_mean" in meta and "log_f0_std" in meta and meta["log_f0_std"]:
        lf0 = (lf0 - meta["log_f0_mean"]) / meta["log_f0_std"]
    vals["log_f0"] = lf0
    unknown = [k for k in meta["cond_order"] if k not in vals]
    return np.array([vals.get(k, 0.0) for k in meta["cond_order"]], dtype=np.float32), unknown


class Decoder:
    def __init__(self, path, latent_dim, cond_dim):
        self.sess = ort.InferenceSession(str(path), providers=["CPUExecutionProvider"])
        self.inputs = self.sess.get_inputs()
        self.L, self.C = latent_dim, cond_dim
        self.out_name = self.sess.get_outputs()[0].name

        def per_item(shape):
            dims = [d for d in shape[1:]] if len(shape) > 1 else list(shape)
            if any(not isinstance(d, int) or d <= 0 for d in dims):
                return None
            return int(np.prod(dims)) if dims else None

        if len(self.inputs) == 2:
            sizes = [per_item(i.shape) for i in self.inputs]
            names = [i.name.lower() for i in self.inputs]
            zi = 0
            if latent_dim != cond_dim and None not in sizes:
                zi = 0 if sizes[0] == latent_dim else 1
            elif ("cond" in names[0] or "label" in names[0]) and not ("cond" in names[1] or "label" in names[1]):
                zi = 1
            self.z_in, self.c_in = self.inputs[zi], self.inputs[1 - zi]
            self.layout = "two"
        elif len(self.inputs) == 1:
            self.layout = "concat"
        else:
            raise SystemExit(f"expected 1 or 2 inputs, got {len(self.inputs)}")

    def describe(self):
        ins = ", ".join(f"{i.name}{list(i.shape)}:{i.type}" for i in self.inputs)
        outs = ", ".join(f"{o.name}{list(o.shape)}:{o.type}" for o in self.sess.get_outputs())
        return f"inputs: {ins}\noutputs: {outs}\nlayout: {self.layout}"

    def __call__(self, z, cond):
        z = np.asarray(z, np.float32).reshape(1, -1)
        cond = np.asarray(cond, np.float32).reshape(1, -1)
        if self.layout == "two":
            feeds = {self.z_in.name: z, self.c_in.name: cond}
        else:
            feeds = {self.inputs[0].name: np.concatenate([z, cond], axis=1)}
        return self.sess.run([self.out_name], feeds)[0].reshape(-1)


def harmonic_mags(x, n_harm=64):
    spec = np.abs(np.fft.rfft(x - x.mean())) / len(x)
    return spec[1 : n_harm + 1]


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("model")
    ap.add_argument("--meta", default=None, help="sid_vae_meta.json (default: next to model, then one level up)")
    ap.add_argument("--a", default="tri", choices=WAVES)
    ap.add_argument("--b", default="saw", choices=WAVES)
    ap.add_argument("--f0", type=float, default=220.0)
    ap.add_argument("--pw", type=float, default=0.5)
    ap.add_argument("--vel", type=float, default=1.0)
    ap.add_argument("--z", type=float, nargs="*", default=None, help="latent vector (default zeros)")
    ap.add_argument("--steps", type=int, default=21)
    ap.add_argument("--plot", default=None, help="save a PNG of the morph (needs matplotlib)")
    args = ap.parse_args()

    model = Path(args.model)
    meta_path = args.meta
    if meta_path is None:
        for cand in (model.parent / "sid_vae_meta.json", model.parent.parent / "sid_vae_meta.json"):
            if cand.exists():
                meta_path = cand
                break
    meta = load_meta(meta_path)
    print(f"meta: {meta_path or '(defaults)'} -> latent_dim={meta['latent_dim']} cond_dim={meta['cond_dim']}")
    print(f"cond_order: {meta['cond_order']}")

    dec = Decoder(model, meta["latent_dim"], meta["cond_dim"])
    print(dec.describe())

    z = np.zeros(meta["latent_dim"], np.float32)
    if args.z:
        z[: len(args.z)] = args.z[: meta["latent_dim"]]

    ts = np.linspace(0.0, 1.0, args.steps)
    a, b = WAVES[args.a], WAVES[args.b]
    tables = []
    for t in ts:
        w = [0.0, 0.0, 0.0]
        w[a] += 1.0 - t
        w[b] += t
        cond, unknown = build_cond(meta, args.pw, args.f0, args.vel, w)
        tables.append(dec(z, cond))
    if unknown:
        print(f"WARNING: cond names the plugin doesn't fill (set to 0): {unknown}")

    tables = np.stack(tables)
    print(f"output length: {tables.shape[1]} samples, range [{tables.min():.3f}, {tables.max():.3f}]")
    if not np.isfinite(tables).all():
        sys.exit("model produced NaN/Inf")

    d0, d1 = tables[0], tables[-1]
    span = np.linalg.norm(d1 - d0) + 1e-12
    h0, h1 = harmonic_mags(d0), harmonic_mags(d1)
    hspan = np.linalg.norm(h1 - h0) + 1e-12

    print(f"\nmorph {args.a} -> {args.b} at f0={args.f0} Hz, pw={args.pw}, z={z.tolist()}")
    print("   t    nonlinearity  spectral")
    for t, d in zip(ts, tables):
        lin = (1 - t) * d0 + t * d1
        hlin = (1 - t) * h0 + t * h1
        nl = np.linalg.norm(d - lin) / span
        sp = np.linalg.norm(harmonic_mags(d) - hlin) / hspan
        print(f"  {t:4.2f}  {nl:10.3f}  {sp:9.3f}")

    steps = np.linalg.norm(np.diff(tables, axis=0), axis=1)
    med = np.median(steps) + 1e-12
    worst = int(np.argmax(steps))
    print(f"\nstep size: median {med:.4f}, max {steps.max():.4f} "
          f"({steps.max() / med:.1f}x median, between t={ts[worst]:.2f} and t={ts[worst + 1]:.2f})")

    if args.plot:
        try:
            import matplotlib
            matplotlib.use("Agg")
            import matplotlib.pyplot as plt
        except ImportError:
            sys.exit("matplotlib not installed; pip install matplotlib")
        fig, axes = plt.subplots(2, 1, figsize=(10, 7))
        idx = np.linspace(0, len(ts) - 1, 6).astype(int)
        x = np.arange(tables.shape[1]) / tables.shape[1]
        for i in idx:
            axes[0].plot(x, tables[i], label=f"t={ts[i]:.2f}", lw=1.2)
        axes[0].set_title(f"decoded cycle, {args.a} -> {args.b}")
        axes[0].legend(fontsize=8)
        axes[1].plot(ts[1:], steps, marker="o")
        axes[1].set_title("step size between neighbouring morph positions")
        axes[1].set_xlabel("t")
        fig.tight_layout()
        fig.savefig(args.plot, dpi=130)
        print(f"plot saved to {args.plot}")


if __name__ == "__main__":
    main()
